# Version-pin strategy:

# =============================================================================
# SYSTEM DEPENDENCIES  (OS binaries — NOT installed by pip)
# Install once via your OS package manager before running video/slide pipelines.
#
#   ffmpeg    — video/audio encode + compose  (video_pipeline_local.py, video_composer.py)
#   ffprobe   — duration lookups             (ships with ffmpeg)
#   pdftoppm  — PDF → PNG rasterisation      (part of poppler, required by pdf2image)
#   soffice   — PPTX → PDF conversion        (LibreOffice; no pip equivalent)
#
# macOS  :  brew install ffmpeg poppler && brew install --cask libreoffice
# Ubuntu :  sudo apt install ffmpeg poppler-utils libreoffice
# Windows:  choco install ffmpeg poppler libreoffice
#
# Verify: ffmpeg -version && ffprobe -version && pdftoppm -v && soffice --version
# =============================================================================
